SELECT Nombre, salario, Departamento_ID FROM empleados

	SELECT MAX(salario) FROM empleados

)

7.RETO PROPIO:
Muestra el id, nombre, y cantidad de empleados bajo el mando de cada jefe


SELECT 
  Jefe_ID,
  (SELECT Nombre FROM empleados
  WHERE empleado_ID = e.Jefe_ID) AS Jefes,
  COUNT(*) "Trabajadores" 
FROM empleados AS e
WHERE Jefe_ID IS NOT NULL 
GROUP BY	Jefes
ORDER BY Trabajadores DESC
