FROM empleados

6. Decena de departamentos
SELECT ROUND(AVG(salario)) "AVG salario" 
FROM empleados

7. Media de salarios por departamento
SELECT ROUND(AVG(salario)) AS media FROM empleados
GROUP BY departamento_ID
ORDER BY media DESC 


8. Con código
SELECT ROUND(AVG(salario)) AS media, departamento_ID
FROM empleados
GROUP BY departamento_ID
ORDER BY media DESC 


9. Info departamentos
SELECT 
  departamento_ID "deptID",
  ROUND(AVG(salario)) AS avgSalario,
  COUNT(*) AS numEmps
FROM empleados
WHERE comision IS NULL OR comision = 0
GROUP BY departamento_ID
ORDER BY numEmps DESC 


10. Media superior a 2000
SELECT departamento_ID "deptID", ROUND(AVG(salario)) "media salarial" FROM empleados
GROUP BY departamento_ID
HAVING AVG(salario) > 2000
ORDER BY AVG(salario) DESC 


11. Departamentos con más de dos personas
SELECT 
  departamento_ID "deptID", 
  ROUND(AVG(salario)) "media salarial",
  COUNT(*) "Empleados #"
FROM empleados
GROUP BY departamento_ID
HAVING COUNT(*) > 2
ORDER BY COUNT(*) DESC 


12. Empleados de igual salario por dept
SELECT departamento_ID, salario, COUNT(salario) "Empleados" FROM empleados 
GROUP BY departamento_ID, salario
-- HAVING COUNT(salario) != 1 -- para ver solo los salarios repetidos


13. Empleados de mismo trabajo por dept
SELECT departamento_ID, Trabajo_ID, COUNT(*) "Empleados" FROM empleados
GROUP BY departamento_ID, Trabajo_ID
HAVING COUNT(*) > 1 -- Varias personas en el mismo trabajo


14. Departamentos con comisión
SELECT departamento_ID, COUNT(comision) "Empleados con comision" FROM empleados
GROUP BY departamento_ID
HAVING COUNT(comision) > 2


15. Empleados con más de 2000

20. Salario medio alto

GROUP BY departamento_ID

23. Media de salarios de algunos empleados
SELECT departamento_ID, ROUND(AVG(salario)) "Salario medio" FROM empleados
WHERE departamento_ID != 13 AND trabajo_ID NOT IN (669, 670) 
GROUP BY departamento_ID
